import time, tracemalloc

def sequential_search(data: list, target: int) -> dict:
    """
Pseudocode (Lis Utari):
i ← 0
ketemu ← false
while (not ketemu) and (i < N) do
if data[i] == target then ketemu ← true
else i ← i + 1
return i if ketemu else -1

T_min = 1 | T_max = n | T_avg = (n+1)/2 | O(n)
    """
    tracemalloc.start()
    t_start = time.perf_counter()

    i, ketemu = 0, False

    while (not ketemu) and (i < len(data)):
        if data[i] == target: ketemu = True
        else: i += 1

    t_end = time.perf_counter()
    cur_mem, peak_mem = tracemalloc.get_traced_memory()
    tracemalloc.stop()

    return {
        "index": i if ketemu else -1,
        "found": ketemu,
        "time_ms": (t_end - t_start) * 1000,
        "memory_kb": peak_mem / 1024,
    }

def selection_sort(data_in: list) -> dict:
    """
Pseudocode (Lis Utari):
for i ← 1 to n-1 do
imin ← i
for j ← i+1 to n do
if a[j] < a[imin] then imin ← j
swap(a[i], a[imin])

Cmp: n(n-1)/2 (SAMA untuk best & worst) | Swap: ≤ n-1 | O(n²)
    """
    data = data_in[:]
    n = len(data)
    cmp, swap = 0, 0

    for i in range(n - 1):
        imin = i
        for j in range(i + 1, n):
            cmp += 1
            if data[j] < data[imin]: imin = j
        if imin != i:
            data[i], data[imin] = data[imin], data[i]
            swap += 1

    return data, cmp, swap

def bubble_sort(data_in: list) -> dict:
    """
Pseudocode (Lis Utari):
for i ← n-1 downto 1 do
for j ← 1 to i do
if a[j+1] < a[j] then swap(a[j], a[j+1])

Cmp: n(n-1)/2 | Swap_min=0 Swap_max=n(n-1)/2 | O(n²)
⚠ LEBIH BOROS swap dibanding Selection Sort!
    """
    data = data_in[:]
    n = len(data)
    cmp, swap = 0, 0

    for i in range(n - 1, 0, -1): # i = n-1 downto 1
        for j in range(1, i + 1): # j = 1 to i
            cmp += 1
            if data[j] < data[j - 1]:
                data[j], data[j-1] = data[j-1], data[j]
                swap += 1

    return data, cmp, swap
